Contexte : addon World of Warcraft Classic "WarLegendRaidTool" (Lua 5.1 de WoW).
Règles du projet (extrait de AGENTS.md) :
- Lua 5.1 : pas de goto, pas de //, pas d'opérateurs bit à bit, pas de table.unpack (utiliser unpack).
- Ne jamais inventer d'API WoW ; une API incertaine est gardée : if C_X and C_X.Y then ... end.
- Les frames sécurisées / actions protégées ne peuvent pas changer en combat : tester InCombatLockdown() d'abord.
- Chaque fichier : local W = WLRT ; local L = W.L ; local db. Données : W.OnReady(function() db = W.DB("cle", { defauts scalaires }) end).
- Événements : W.On("EVENT", fn, always, inactive) renvoie un handler ; inactive = true le démarre éteint, W.Activate(handler, on) l'allume.
  always = true : s'exécute même si le joueur n'est pas dans la guilde (réservé au cœur de l'addon). Sinon les handlers ne tournent que si W.Allowed().
- Textes visibles : L("English text") est un APPEL de fonction, jamais L["..."]. Le français va dans W.AddFR({...}) du même fichier.
  La langue n'est connue qu'à ADDON_LOADED : appeler L() dans les fonctions de construction / handlers / W.OnReady, pas au chargement du fichier.
- Nouvelle commande /wl : W.slash.nom dans le module, + liste HELP de Core.lua (avec texte français), + tableau "Commandes" de README.md.
- Outils : node tools/check.js (vérifications), bash tools/sync.sh (copie vers le dossier du jeu). Pas de bump de version sans demande.

  62: ---------------------------------------------------------------------------
  63: -- Cost of our own work (/wl perf): every event handler, timer, roster and ready function is timed
  64: -- (debugprofilestop, microseconds) and the slowest are listed with the file and line that made them.
  65: ---------------------------------------------------------------------------
  66: local clock = debugprofilestop
  67: local costs = {} -- [label] = { max, count, total }
  68: function W.Where(level)
  69:     local s = debugstack and debugstack((level or 2) + 1, 1, 0) or ""
  70:     local file, line = s:match("([%w_]+%.lua)%D+(%d+)")
  71:     return file and (file .. ":" .. line) or "?"
  72: end
  73: 
  74: function W.Measure(label, fn, ...)
  75:     if not clock then return pcall(fn, ...) end
  76:     local t0 = clock()
  77:     local ok, err = pcall(fn, ...)
  78:     local dt = clock() - t0
  79:     local c = costs[label]
  80:     if not c then c = { max = 0, count = 0, total = 0 }; costs[label] = c end
  81:     c.count, c.total = c.count + 1, c.total + dt
  82:     if dt > c.max then c.max = dt end
  83:     return ok, err
  84: end
  85: 
  86: -- the n slowest (one run), { label, max ms, count, total ms }
  87: function W.SlowestWork(n)
  88:     local list = {}
  89:     for label, c in pairs(costs) do list[#list + 1] = { label, c.max, c.count, c.total } end
  90:     table.sort(list, function(a, b) return a[2] > b[2] end)
  91:     while #list > (n or 5) do table.remove(list) end
  92:     return list
  93: end
  94: 
  95: -- the n busiest (most time in all since the login), same entries: what costs frames, even in small runs
  96: function W.BusiestWork(n)
  97:     local list = {}
  98:     for label, c in pairs(costs) do list[#list + 1] = { label, c.max, c.count, c.total } end
  99:     table.sort(list, function(a, b) return a[4] > b[4] end)
 100:     while #list > (n or 5) do table.remove(list) end
 101:     return list
 102: end

 340: -- Loads the raid addon now. Handlers its files registered after the login / roster events already

 343:     if W.raidLoaded then return true end
 344:     if loadFailed or not (C_AddOns and C_AddOns.LoadAddOn) then return false end

 346:     local called, ok = W.Measure("load the raid modules", C_AddOns.LoadAddOn, RAID_ADDON)

 349:     late = nil

 403: function Dispatch(prefix, msg, channel, sender)
 404:     local fn = addonHandlers[prefix]
 405:     if not fn then
 406:         -- a message for a raid module that is not loaded yet: load it, then deliver the message.
 407:         -- Only from my group / guild (a whisper of a stranger must not load anything), and after the fight.
 408:         if lodPrefix[prefix] and not W.raidLoaded and not loadFailed and W.access == true then
 409:             local who = W.Safe(sender)
 410:             if type(who) ~= "string" or who == "" then return end
 411:             if not GROUP_CHANNELS[channel] then
 412:                 if channel ~= "WHISPER" then return end
 413:                 local full = W.FullName(who)
 414:                 if not (W.UnitByName(full) or W.GuildMember(full)) then return end
 415:             end
 416:             if #pendingMsgs < 50 then pendingMsgs[#pendingMsgs + 1] = { prefix, msg, channel, sender } end
 417:             if InCombatLockdown() then waitCombat = true; return end -- loaded after the fight, messages kept
 418:             W.LoadRaid()
 419:         end
 420:         return
 421:     end
 422:     if offFiles[addonFiles[prefix]] then return end
 423:     msg, sender = W.Safe(msg), W.Safe(sender)
 424:     if not msg or not sender or sender == "" then return end
 425:     local full = W.FullName(sender)

 427:         if channel ~= "WHISPER" then return end
 428:         if not (W.UnitByName(full) or W.GuildMember(full)) then return end
 429:     end
 430:     fn(msg, W.BaseName(sender), channel, full)

 533: -- Saved data format versions. A module changing its saved format registers a migration
 534: -- (W.Migration(version, fn)); they run once each, in order, before the modules start.
 535: -- A failing migration resets only the data it was working on, never the whole file.
 536: local migrations = {}
 537: function W.Migration(version, key, fn)
 538:     migrations[#migrations + 1] = { v = version, key = key, fn = fn }
 539: end
 540: 
 541: local function Migrate(db)
 542:     local from = tonumber(db.schema) or 0
 543:     local top = from
 544:     table.sort(migrations, function(a, b) return a.v < b.v end)
 545:     for _, m in ipairs(migrations) do
 546:         if m.v > from then
 547:             local ok, err = pcall(m.fn, db)
 548:             if not ok then
 549:                 geterrorhandler()(err)
 550:                 if m.key then db[m.key] = nil end -- that module starts clean, the rest is kept
 551:             end
 552:         end
 553:         if m.v > top then top = m.v end
 554:     end
 555:     db.schema = top
 556: end

 705: -- Addon message transport. The modern API returns a result code where only Success (0) means the
 706: -- message was accepted. Returns "sent", "throttle" (client limit: retry in a moment), "locked"
 707: -- (AddOnMessageLockdown, e.g. during an encounter: retry after it) or "failed" (not in a group,

 710: local AR_OK = AR.Success or 0

 712: local AR_LOCKED = AR.AddOnMessageLockdown or 11

 714: function W.SendRaw(prefix, msg, channel, target)
 715:     if not channel then return "failed" end
 716:     -- WHISPER: an answer to one player (Readiness), only with a target
 717:     if channel == "WHISPER" then
 718:         if not target or target == "" then return "failed" end
 719:     elseif channel ~= "RAID" and channel ~= "PARTY" and channel ~= "GUILD" and channel ~= "INSTANCE_CHAT" then
 720:         return "failed"
 721:     end
 722:     if C_ChatInfo and C_ChatInfo.SendAddonMessage then
 723:         local ok, r = pcall(C_ChatInfo.SendAddonMessage, prefix, msg, channel, target)
 724:         if not ok then return "failed" end
 725:         if r == nil or r == true or r == AR_OK then return "sent" end
 726:         if r == AR_LOCKED then return "locked" end
 727:         if AR_THROTTLE[r] then return "throttle" end
 728:         return "failed"
 729:     elseif SendAddonMessage then
 730:         SendAddonMessage(prefix, msg, channel, target)
 731:         return "sent"
 732:     end
 733:     return "failed"
 734: end

1962: local ACCENTS = { ["é"] = "e", ["è"] = "e", ["ê"] = "e", ["ë"] = "e", ["à"] = "a", ["â"] = "a", ["ä"] = "a",

1965: local function Plain(s)
1966:     s = (s or ""):gsub("|c%x%x%x%x%x%x%x%x", ""):gsub("|r", ""):gsub("|T.-|t", "")

1969: local function Norm(s)
1970:     s = Plain(s)

1977: -- The tabs not opened yet are built one at a time in the background (one every 0.1 s, never in combat), so
1978: -- the search box knows their options without one long freeze at the first search.
1979: local slowIndexing = false
1980: function W.IndexTabsSlowly()
1981:     if slowIndexing or allIndexed or not main then return end
1982:     slowIndexing = true
1983:     local i = 0
1984:     local function step()
1985:         if allIndexed then slowIndexing = false; return end
1986:         if InCombatLockdown() then W.After(1, step); return end
1987:         -- not in a battleground: the frames load there already, the tabs are built once outside (a first
1988:         -- search still builds what is missing, see W.SearchOptions)
1989:         local inInstance, kind = IsInInstance()
1990:         if inInstance and kind == "pvp" then W.After(10, step); return end
1991:         i = i + 1
1992:         local m = W.modules[i]
1993:         if not m then allIndexed = true; slowIndexing = false; return end
1994:         if not m.panel and not NOT_BUILT[m.key] then W.EnsurePanel(m) end
1995:         W.After(0.1, step)
1996:     end
1997:     W.After(1, step)
1998: end

2000: function W.SearchOptions(query, max)
2001:     local q = Norm(query):gsub("^%s+", ""):gsub("%s+$", "")
2002:     if #q < 2 or not main then return {} end
2003:     if not allIndexed then
2004:         allIndexed = true
2005:         for _, m in ipairs(W.modules) do
2006:             if not NOT_BUILT[m.key] then W.EnsurePanel(m) end
2007:         end
2008:     end
2009:     local out, seen = {}, {}
2010:     local function add(m, widget, text)
2011:         local key = m.key .. "|" .. text
2012:         if seen[key] or #out >= max then return end
2013:         seen[key] = true
2014:         out[#out + 1] = { module = m, widget = widget,
2015:             label = "|cffffd100" .. L(m.label) .. "|r" .. (widget and ("  |cff777777>|r  " .. text) or "") }
2016:     end
2017:     for _, m in ipairs(W.modules) do -- the tools first
2018:         if Norm(L(m.label)):find(q, 1, true) then add(m, nil, "") end
2019:     end
2020:     for _, m in ipairs(W.modules) do
2021:         for _, e in ipairs(m.index or {}) do
2022:             local text = Plain(e.text)
2023:             if Norm(text):find(q, 1, true) then add(m, e.widget, text) end
2024:         end
2025:     end
2026:     return out
2027: end
